set(WP_GENERATED "${CMAKE_SOURCE_DIR}/build/wiisports/generated")

file(GLOB WP_DSP_SOURCES CONFIGURE_DEPENDS "${WP_GENERATED}/dsp/*.cpp")
if(WP_DSP_SOURCES)
    target_sources(wp_engine PRIVATE ${WP_DSP_SOURCES})
    set_source_files_properties(${WP_DSP_SOURCES} PROPERTIES COMPILE_OPTIONS -O2)
else()
    target_sources(wp_engine PRIVATE "${CMAKE_SOURCE_DIR}/engine/src/audio/no_dsp_code.cpp")
endif()

file(GLOB WP_DOL_SOURCES CONFIGURE_DEPENDS "${WP_GENERATED}/dol/*.cpp")
file(GLOB_RECURSE WP_MODULE_SOURCES CONFIGURE_DEPENDS "${WP_GENERATED}/modules/*.cpp")
if(NOT WP_DOL_SOURCES)
    message(STATUS "No Wii Sports generated code in ${WP_GENERATED}/dol; run the Wii Sports analysis and recompiler pipeline first")
    return()
endif()

add_library(wiisports_code STATIC game.cpp ${WP_DOL_SOURCES})
set_source_files_properties(${WP_DOL_SOURCES} ${WP_MODULE_SOURCES} PROPERTIES COMPILE_OPTIONS "${WP_CODE_OPT}")
target_include_directories(wiisports_code PUBLIC "${WP_GENERATED}/dol")
if(WP_MODULE_SOURCES)
    target_sources(wiisports_code PRIVATE ${WP_MODULE_SOURCES})
else()
    target_sources(wiisports_code PRIVATE "${CMAKE_SOURCE_DIR}/engine/src/core/no_modules.cpp")
endif()
target_link_libraries(wiisports_code PUBLIC wp_engine)
target_compile_options(wiisports_code PRIVATE -Wall -Wextra -Wno-unused-label -Wno-unused-parameter -Wno-clobbered)

add_executable(wiisports)
if(APPLE)
    target_link_libraries(wiisports PRIVATE wp_app wiisports_code wp_engine)
else()
    target_link_libraries(wiisports PRIVATE wp_app "$<LINK_GROUP:RESCAN,wiisports_code,wp_engine>")
endif()
target_compile_options(wiisports PRIVATE -Wall -Wextra)
if(WIN32 AND NOT WP_CONSOLE)
    set_target_properties(wiisports PROPERTIES WIN32_EXECUTABLE ON)
endif()
if(TARGET SDL3::SDL3)
    add_custom_command(TARGET wiisports POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:SDL3::SDL3> $<TARGET_FILE_DIR:wiisports>)
endif()
